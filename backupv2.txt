import os
import re
from collections import Counter, deque

import cv2
import numpy as np
from ultralytics import YOLO
from paddleocr import PaddleOCR

# --- Stability ---
os.environ["FLAGS_use_mkldnn"] = "0"
os.environ["KMP_DUPLICATE_LIB_OK"] = "TRUE"
os.environ["PADDLE_PDX_DISABLE_MODEL_SOURCE_CHECK"] = "True"

# --- Video configuration ---
VIDEO_INPUT = "sample.mp4"
VIDEO_OUTPUT = "anpr_output.mp4"
YOLO_MODEL = "v3.pt"
FRAME_SKIP = 2  # lower skip = better recall
VEHICLE_CLASS_IDS = {2, 3, 5, 7}  # car, motorcycle, bus, truck in COCO


# ---------- Text postprocessing ----------
def fix_character_confusion(text: str) -> str:
    # Apply replacements mostly on prefix area where letters are expected.
    confusion = {"8": "B", "0": "O", "1": "I", "5": "S", "2": "Z", "6": "G"}
    return "".join(confusion.get(c, c) for c in text)


def clean_sl_text(raw_str: str) -> str:
    text = re.sub(r"[^A-Z0-9]", "", raw_str.upper())
    digits_found = re.findall(r"\d{4}", text)
    if not digits_found:
        return "NOT_FOUND"

    num_part = digits_found[-1]
    prefix_raw = text[: text.find(num_part)]
    prefix_raw = fix_character_confusion(prefix_raw)

    noise = [
        "SRILANKA",
        "SRI",
        "LANKA",
        "WP",
        "CP",
        "SP",
        "NW",
        "NC",
        "UP",
        "SG",
        "VA",
        "EP",
        "NP",
    ]
    for n in noise:
        prefix_raw = prefix_raw.replace(n, "")

    letters = "".join(re.findall(r"[A-Z]", prefix_raw))
    if len(letters) == 0:
        return "NOT_FOUND"
    if len(letters) > 3:
        letters = letters[-3:]

    return f"{letters}-{num_part}"


# ---------- Orientation helpers ----------
def get_rotation_code(cap: cv2.VideoCapture):
    """Return cv2 rotate code from orientation metadata if available."""
    # OpenCV can expose metadata in degrees for some files/codecs.
    orientation = cap.get(cv2.CAP_PROP_ORIENTATION_META)
    if orientation == 90:
        return cv2.ROTATE_90_CLOCKWISE
    if orientation == 180:
        return cv2.ROTATE_180
    if orientation == 270:
        return cv2.ROTATE_90_COUNTERCLOCKWISE
    return None


def maybe_rotate(frame: np.ndarray, rotate_code):
    if rotate_code is None:
        return frame
    return cv2.rotate(frame, rotate_code)


# ---------- Plate region proposals ----------
def plate_candidates(vehicle_crop: np.ndarray):
    """Generate likely plate ROIs from a vehicle crop using contour heuristics."""
    if vehicle_crop.size == 0:
        return []

    h, w = vehicle_crop.shape[:2]
    gray = cv2.cvtColor(vehicle_crop, cv2.COLOR_BGR2GRAY)
    blur = cv2.bilateralFilter(gray, 9, 75, 75)
    edges = cv2.Canny(blur, 50, 150)

    contours, _ = cv2.findContours(edges, cv2.RETR_TREE, cv2.CHAIN_APPROX_SIMPLE)
    candidates = []

    for c in contours:
        x, y, cw, ch = cv2.boundingRect(c)
        if cw == 0 or ch == 0:
            continue

        aspect = cw / float(ch)
        area_ratio = (cw * ch) / float(w * h + 1e-6)

        # Common plate geometry limits.
        if 2.0 <= aspect <= 6.5 and 0.01 <= area_ratio <= 0.25:
            # Add small border for OCR context.
            pad_x, pad_y = max(4, cw // 12), max(4, ch // 6)
            x1 = max(0, x - pad_x)
            y1 = max(0, y - pad_y)
            x2 = min(w, x + cw + pad_x)
            y2 = min(h, y + ch + pad_y)
            roi = vehicle_crop[y1:y2, x1:x2]
            if roi.size > 0:
                candidates.append(roi)

    # fallback: lower-center strip (where rear/front plates often appear)
    if not candidates:
        y1, y2 = int(h * 0.50), int(h * 0.92)
        x1, x2 = int(w * 0.10), int(w * 0.95)
        fallback = vehicle_crop[y1:y2, x1:x2]
        if fallback.size > 0:
            candidates.append(fallback)

    # keep top N by area
    candidates = sorted(candidates, key=lambda im: im.shape[0] * im.shape[1], reverse=True)
    return candidates[:5]


# ---------- OCR helpers ----------
def ocr_variants(img: np.ndarray):
    """Create multiple preprocess variants to improve OCR recall."""
    gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
    clahe = cv2.createCLAHE(clipLimit=2.5, tileGridSize=(8, 8)).apply(gray)
    up = cv2.resize(clahe, None, fx=2.0, fy=2.0, interpolation=cv2.INTER_CUBIC)

    th_adapt = cv2.adaptiveThreshold(
        up,
        255,
        cv2.ADAPTIVE_THRESH_GAUSSIAN_C,
        cv2.THRESH_BINARY,
        31,
        5,
    )

    _, th_otsu = cv2.threshold(up, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU)

    return [
        cv2.cvtColor(up, cv2.COLOR_GRAY2BGR),
        cv2.cvtColor(th_adapt, cv2.COLOR_GRAY2BGR),
        cv2.cvtColor(th_otsu, cv2.COLOR_GRAY2BGR),
    ]


def read_plate_text(ocr: PaddleOCR, roi: np.ndarray):
    best = ("NOT_FOUND", 0.0)
    for variant in ocr_variants(roi):
        ocr_res = ocr.ocr(variant, cls=True)
        if not ocr_res or not ocr_res[0]:
            continue

        raw = "".join([seg[1][0] for seg in ocr_res[0]])
        conf = float(np.mean([seg[1][1] for seg in ocr_res[0]]))
        candidate = clean_sl_text(raw)

        if candidate != "NOT_FOUND" and conf > best[1]:
            best = (candidate, conf)

    return best


def main():
    print("Loading models...")
    model = YOLO(YOLO_MODEL)
    ocr = PaddleOCR(lang="en", show_log=False, use_angle_cls=True)

    cap = cv2.VideoCapture(VIDEO_INPUT)
    if not cap.isOpened():
        raise FileNotFoundError(f"Could not open input video: {VIDEO_INPUT}")

    rotate_code = get_rotation_code(cap)

    fps = cap.get(cv2.CAP_PROP_FPS) or 25.0
    in_w = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))
    in_h = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))
    total_frames = int(cap.get(cv2.CAP_PROP_FRAME_COUNT))

    out_w, out_h = (in_h, in_w) if rotate_code in (cv2.ROTATE_90_CLOCKWISE, cv2.ROTATE_90_COUNTERCLOCKWISE) else (in_w, in_h)

    fourcc = cv2.VideoWriter_fourcc(*"mp4v")
    out = cv2.VideoWriter(VIDEO_OUTPUT, fourcc, fps, (out_w, out_h))

    # basic temporal smoothing via nearest-center matching
    track_memory = {}  # id -> {center, texts deque}
    next_id = 0

    frame_count = 0
    while cap.isOpened():
        ok, frame = cap.read()
        if not ok:
            break

        frame = maybe_rotate(frame, rotate_code)
        h, w = frame.shape[:2]
        frame_count += 1

        current_boxes = []

        if frame_count % FRAME_SKIP == 0:
            results = model.predict(source=frame, conf=0.18, iou=0.5, verbose=False)
            for r in results:
                for box in r.boxes:
                    cls_id = int(box.cls[0]) if box.cls is not None else -1
                    if cls_id not in VEHICLE_CLASS_IDS:
                        continue

                    x1, y1, x2, y2 = map(int, box.xyxy[0])
                    x1 = max(0, x1)
                    y1 = max(0, y1)
                    x2 = min(w, x2)
                    y2 = min(h, y2)
                    if x2 <= x1 or y2 <= y1:
                        continue

                    vehicle_crop = frame[y1:y2, x1:x2]
                    best_plate, best_conf = "NOT_FOUND", 0.0
                    for roi in plate_candidates(vehicle_crop):
                        plate, conf = read_plate_text(ocr, roi)
                        if conf > best_conf:
                            best_plate, best_conf = plate, conf

                    current_boxes.append((x1, y1, x2, y2, best_plate, best_conf))

            # update memory
            for x1, y1, x2, y2, plate, _conf in current_boxes:
                cx, cy = (x1 + x2) // 2, (y1 + y2) // 2
                matched = None
                min_d = 10**9
                for tid, data in track_memory.items():
                    px, py = data["center"]
                    d = abs(px - cx) + abs(py - cy)
                    if d < min_d and d < 120:
                        min_d = d
                        matched = tid

                if matched is None:
                    matched = next_id
                    next_id += 1
                    track_memory[matched] = {"center": (cx, cy), "texts": deque(maxlen=8), "box": (x1, y1, x2, y2)}

                track_memory[matched]["center"] = (cx, cy)
                track_memory[matched]["box"] = (x1, y1, x2, y2)
                if plate != "NOT_FOUND":
                    track_memory[matched]["texts"].append(plate)

        # draw
        for data in track_memory.values():
            x1, y1, x2, y2 = data["box"]
            cv2.rectangle(frame, (x1, y1), (x2, y2), (0, 255, 0), 2)

            if data["texts"]:
                stable_text = Counter(data["texts"]).most_common(1)[0][0]
            else:
                stable_text = "PLATE_NOT_CLEAR"

            font = cv2.FONT_HERSHEY_SIMPLEX
            scale, thickness = 0.7, 2
            (tw, th), _ = cv2.getTextSize(stable_text, font, scale, thickness)
            by1 = max(0, y1 - th - 10)
            by2 = y1
            cv2.rectangle(frame, (x1, by1), (x1 + tw + 10, by2), (0, 255, 0), -1)
            cv2.putText(frame, stable_text, (x1 + 5, by2 - 4), font, scale, (0, 0, 0), thickness)

        out.write(frame)

        if frame_count % 30 == 0:
            print(f"Processed {frame_count}/{total_frames} frames")

    cap.release()
    out.release()
    print(f"Done. Saved output to {VIDEO_OUTPUT}")


if __name__ == "__main__":
    main()
